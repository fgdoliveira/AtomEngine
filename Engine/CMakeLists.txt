# PUBLIC: what the engine's own headers expose (SDL types, GLM maths).
# v0.0.19: no ImGui - the development tools are the workspace's
# (Tools/Workspace); the F1 overlay draws with the engine's own UI.
target_link_libraries(
    AtomEngine
    PUBLIC
        SDL3::SDL3
        glm::glm
        AtomProfile # M95: the profiling switch (and the built-in tracer, when on) for every user
)

# The F1 overlay's font, embedded so it needs no asset file: ProggyClean
# (MIT, Debug/Fonts/LICENSE.txt) as a byte array, regenerated when it changes.
set(ATOM_OVERLAY_FONT "${CMAKE_CURRENT_SOURCE_DIR}/Debug/Fonts/ProggyClean.ttf")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ATOM_OVERLAY_FONT}")
file(READ "${ATOM_OVERLAY_FONT}" ATOM_OVERLAY_FONT_HEX HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," ATOM_OVERLAY_FONT_BYTES "${ATOM_OVERLAY_FONT_HEX}")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/Generated/OverlayFont.cpp" CONTENT
"// Generated by Engine/CMakeLists.txt from Debug/Fonts/ProggyClean.ttf - do not edit.
#include \"Debug/PerfOverlay.h\"
namespace
{
    const unsigned char Bytes[] = { ${ATOM_OVERLAY_FONT_BYTES} };
}
std::span<const unsigned char> Atom::PerfOverlay::FontData() { return Bytes; }
")

# M101: process and GPU memory (psapi, DXGI) and crash minidumps (dbghelp),
# all part of Windows.
if(WIN32)
    target_link_libraries(AtomEngine PRIVATE psapi dxgi dbghelp)
endif()

# The GLM settings change what GLM's headers mean, so every user needs
# them, and so does ATOM_DEV_TOOLS (Core/DevSwitch.h compiles differently
# without it: a game must see the value the engine was built with, M127);
# the version string is the engine's own.
target_compile_definitions(
    AtomEngine
    PUBLIC
        GLM_FORCE_DEPTH_ZERO_TO_ONE
        GLM_FORCE_RADIANS
        ATOM_DEV_TOOLS=${ATOM_DEV_TOOLS_VALUE}
    PRIVATE
        ATOM_VERSION="${PROJECT_VERSION}"
)

target_compile_features(
    AtomEngine
    PUBLIC
        cxx_std_20
)
